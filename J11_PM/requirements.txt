# ===== Runtime : ce qui fait tourner ton API =====
fastapi                 # le framework web
uvicorn[standard]       # le serveur qui lance ton app (uvicorn app.main:app)
sqlalchemy              # l'ORM (← J10)
psycopg[binary]         # le driver PostgreSQL v3 (← J10)
alembic                 # les migrations de base (← J10)
pydantic-settings       # lecture de la config depuis .env (secrets, ← J11)
python-dotenv           # support du fichier .env (utilisé par pydantic-settings)
email-validator         # requis si un schéma utilise EmailStr (User a un champ email)
python-multipart        # requis par le login OAuth2 password flow (← J11)
PyJWT                   # les tokens JWT (← J11) — PAS le paquet "jwt" !
pwdlib[argon2]          # hachage des mots de passe avec Argon2 (← J10-J11)
slowapi                 # le rate limiting (← J11, ton core/limiter.py)

# ===== Dev / tests : pas nécessaire en prod =====
httpx                   # requis par TestClient (← J7/J9)
pytest                  # le lanceur de tests (← J9)
pytest-cov              # la couverture (aujourd'hui)
ruff                    # linter/formatter (← J7)
mypy                    # vérif de types (← J3)
